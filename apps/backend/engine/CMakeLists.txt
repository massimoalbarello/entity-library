cmake_minimum_required(VERSION 3.24)
project(entity_library LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../../..)
find_package(OpenCV REQUIRED COMPONENTS core imgcodecs imgproc)
set(CLIP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(CLIP_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CLIP_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(${ROOT}/vendor/clip.cpp ${CMAKE_BINARY_DIR}/clip)
add_library(sqlite STATIC ${ROOT}/vendor/sqlite3.c)
target_compile_definitions(sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_DQS=0 SQLITE_DEFAULT_MEMSTATUS=0 SQLITE_OMIT_LOAD_EXTENSION)
file(READ ${CMAKE_CURRENT_SOURCE_DIR}/../src/models/manifest.json MANIFEST)
configure_file(src/model_manifest.h.in generated/model_manifest.h @ONLY)
add_executable(entity-core src/main.cpp)
target_include_directories(entity-core PRIVATE ${ROOT}/vendor ${ROOT}/vendor/clip.cpp/ggml/include ${CMAKE_BINARY_DIR}/generated)
target_link_libraries(entity-core PRIVATE clip sqlite ${OpenCV_LIBS})
if(UNIX AND NOT APPLE)
 target_link_libraries(entity-core PRIVATE pthread dl)
 # musl defaults to small worker stacks; GGML graph construction needs more.
 # This reserves address space, not 8 MiB of resident memory per thread.
 target_link_options(entity-core PRIVATE "-Wl,-z,stack-size=8388608")
endif()
